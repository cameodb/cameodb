// storage/src/lib.rs

struct PreparedOp {
    wal_bytes: Vec<u8>,
    doc_bytes: Option<Vec<u8>>,
    id: String,
}

impl HybridStore {
    pub fn apply_batch(&self, index: &str, ops: Vec<WalOp>) -> Result<(Vec<u64>, usize), StoreError> {
        // Step 1: Serialize outside of lock (CPU work)
        let mut prepared_ops = Vec::with_capacity(ops.len());
        for op in ops {
            match op {
                WalOp::Put { id, json_blob } => {
                    let wal_bytes = serde_json::to_vec(&op)?;
                    let doc_bytes = json_blob.map(|blob| 
                        serde_json::to_vec(&StoredDoc { json_blob: Some(&blob) })?
                    );
                    prepared_ops.push(PreparedOp { wal_bytes, doc_bytes, id });
                }
                WalOp::Delete { id } => {
                    prepared_ops.push(PreparedOp { 
                        wal_bytes: serde_json::to_vec(&op)?, 
                        doc_bytes: None, 
                        id 
                    });
                }
            }
        }
        
        // Step 2: Minimal critical section (IO work only)
        let mut write_txn = self.kv.begin_write()?;
        let mut new_docs = 0;
        
        {
            let mut wal_table = write_txn.open_table(wal_table_def)?;
            let mut data_table = write_txn.open_table(data_table_def)?;
            
            for (seq_id, prepared) in prepared_ops.into_iter().zip(start_seq..) {
                wal_table.insert(seq_id, prepared.wal_bytes.as_slice())?;
                
                if let Some(doc_bytes) = prepared.doc_bytes {
                    if data_table.insert(&prepared.id, doc_bytes.as_slice())?.is_none() {
                        new_docs += 1;
                    }
                } else {
                    data_table.remove(&prepared.id)?;
                }
            }
        }
        
        write_txn.commit()?;
        Ok((seq_ids, new_docs))
    }
}

pub struct MicroshardActor {
    shard_id: Uuid,
    store: Option<Arc<HybridStore>>,        // For Reads (Parallel)
    writer_tx: Option<flume::Sender<StorageCommand>>, // For Writes (Serialized)
    storage_config: StorageConfig,
    writer_handle: Option<std::thread::JoinHandle<()>>, // Thread management
}

impl MicroshardActor {
    /// Performance-optimized startup with dedicated writer thread
    pub async fn start(&mut self) -> Result<(), OrchestratorError> {
        // Initialize Store (Blocking I/O)
        let store = Arc::new(tokio::task::spawn_blocking({
            let cfg = self.storage_config.clone();
            move || HybridStore::new(cfg)
        }).await??);
        
        self.store = Some(store.clone());
        
        // PERFORMANCE: Unbounded channel for maximum throughput
        let (tx, rx) = flume::unbounded();
        self.writer_tx = Some(tx);
        
        // PERFORMANCE: Dedicated writer thread for serialized I/O
        let writer_store = store.clone();
        let shard_id = self.shard_id;
        
        let handle = std::thread::Builder::new()
            .name(format!("writer-shard-{}", shard_id))
            .spawn(move || {
                while let Ok(cmd) = rx.recv() {
                    match cmd {
                        StorageCommand::Write { req, reply } => {
                            let op = WalOp::Put { id: req.id, json_blob: Some(req.doc) };
                            let res = writer_store.apply_write(&req.index, op)
                                .map_err(OrchestratorError::from);
                            let _ = reply.send(res);
                        }
                        StorageCommand::BatchWrite { req, reply } => {
                            // Legacy path for compatibility
                            let ops: Vec<WalOp> = req.docs.into_iter().map(|doc| WalOp::Put {
                                id: doc.id,
                                json_blob: Some(doc.doc),
                            }).collect();
                            
                            let res = writer_store.apply_batch(&req.index, ops)
                                .map(|(seq_ids, _)| seq_ids)
                                .map_err(OrchestratorError::from);
                            let _ = reply.send(res);
                        }
                        // PERFORMANCE CRITICAL: Optimized batch path
                        StorageCommand::PrePackedBatchWrite { index, ops, reply } => {
                            let res = writer_store.apply_pre_packed(&index, ops)
                                .map_err(OrchestratorError::from);
                            let _ = reply.send(res);
                        }
                        StorageCommand::Shutdown(reply) => {
                            let _ = writer_store.shutdown();
                            let _ = reply.send(());
                            break;
                        }
                    }
                }
            })?;
        
        self.writer_handle = Some(handle);
        Ok(())
    }

    // READER PATH: Parallel (Thread Pool) - UNCHANGED
    pub async fn handle_search(&self, req: SearchRequest) -> Result<SearchReply, OrchestratorError> {
        let store = self.store.as_ref().ok_or_else(|| {
            OrchestratorError::Io(std::io::Error::new(
                std::io::ErrorKind::NotFound,
                "HybridStore not initialized",
            ))
        })?.clone();
        
        tokio::task::spawn_blocking(move || {
            // Existing search logic...
            store.search_documents(/* ... */)
        }).await?
        .map_err(OrchestratorError::from)
    }

    // WRITER PATH: Optimized (Dedicated Thread)
    pub async fn handle_write(&self, req: WriteRequest) -> Result<u64, OrchestratorError> {
        let (reply_tx, reply_rx) = tokio::sync::oneshot::channel();
        self.writer_tx.as_ref().ok_or_else(|| {
            OrchestratorError::Io(std::io::Error::new(
                std::io::ErrorKind::NotFound,
                "Writer channel not initialized",
            ))
        })?.send_async(StorageCommand::Write { req, reply: reply_tx }).await
            .map_err(|_| OrchestratorError::Io(std::io::Error::other("Writer closed")))?;
        reply_rx.await
            .map_err(|_| OrchestratorError::Io(std::io::Error::other("Writer dropped response")))?
    }
}

// storage/src/lib.rs - Performance-optimized serialization methods

impl HybridStore {
    /// CPU-only work: serialize and prepare operations with all optimizations
    pub fn prepare_batch_ops(&self, ops: Vec<WalOp>) -> Result<Vec<PrePackedOp>, StoreError> {
        let mut prepared = Vec::with_capacity(ops.len());
        
        for op in ops {
            match op {
                WalOp::Put { id, json_blob } => {
                    // PERFORMANCE: Handle schema evolution and shadow field filtering
                    let filtered_json_blob = json_blob
                        .clone()
                        .map(|blob| filter_shadow_fields_owned(blob, &schema));

                    // PERFORMANCE: Pre-serialize WAL operation
                    let wal_data = serde_json::to_vec(&WalOp::Put {
                        id: id.clone(),
                        json_blob: filtered_json_blob.clone(),
                    })?;

                    // PERFORMANCE: Pre-serialize stored document
                    let doc_data = filtered_json_blob.as_ref().map(|blob| {
                        serde_json::to_vec(&StoredDoc { json_blob: Some(blob) })
                    }).transpose()?;

                    prepared.push(PrePackedOp {
                        wal_data,
                        doc_data,
                        op_type: OpType::Put,
                        id,
                    });
                }
                WalOp::Delete { id } => {
                    prepared.push(PrePackedOp {
                        wal_data: serde_json::to_vec(&WalOp::Delete { id: id.clone() })?,
                        doc_data: None,
                        op_type: OpType::Delete,
                        id,
                    });
                }
            }
        }
        Ok(prepared)
    }
    
    /// I/O-only work: ultra-fast database writes with pre-serialized data
    /// 
    /// NOTE: This method must also handle Tantivy indexing after redb commit.
    /// For Tantivy indexing, deserialize doc_data to extract json_blob:
    /// 
    /// ```rust
    /// if let Some(doc_data) = &prepared_op.doc_data {
    ///     let stored_doc: StoredDoc = serde_json::from_slice(doc_data)?;
    ///     if let Some(json_blob) = stored_doc.json_blob {
    ///         // Use json_blob to build Tantivy document with schema fields
    ///     }
    /// }
    /// ```
    pub fn apply_pre_packed(&self, index: &str, prepared_ops: Vec<PrePackedOp>) -> Result<(Vec<u64>, usize), StoreError> {
        // Get or create the index (for Tantivy writer and fields)
        let (writer_arc, fields) = self.get_or_create_index(index)?;
        
        // Get schema for Tantivy field indexing
        let schema = if let Some(schema) = self.get_schema_cached(index)? {
            schema
        } else {
            self.get_schema(index)?
                .map(Arc::new)
                .unwrap_or_else(|| Arc::new(IndexSchema::default()))
        };
        
        // PERFORMANCE: Atomic sequence ID generation
        let start_seq = self.current_seq.get(index)
            .ok_or_else(|| StoreError::IndexNotFound(format!("Index: {}", index)))?
            .fetch_add(prepared_ops.len() as u64, Ordering::SeqCst) + 1 - prepared_ops.len() as u64;
        
        // PERFORMANCE: Minimal critical section - only database I/O
        let mut write_txn = self.kv.begin_write()?;
        let mut new_documents_count = 0;
        let mut tantivy_ops = Vec::new();
        
        // Create dynamic table definitions
        let data_table_name = format!("data_{}", index);
        let wal_table_name = format!("wal_{}", index);
        let data_table_def = TableDefinition::<&str, &[u8]>::new(&data_table_name);
        let wal_table_def = TableDefinition::<u64, &[u8]>::new(&wal_table_name);
        
        {
            let mut wal_table = write_txn.open_table(wal_table_def)?;
            let mut data_table = write_txn.open_table(data_table_def)?;
            
            // PERFORMANCE: Fast path - just insert pre-serialized bytes
            for (prepared_op, seq_id) in prepared_ops.into_iter().zip(start_seq..) {
                wal_table.insert(seq_id, prepared_op.wal_data.as_slice())?;
                
                match prepared_op.op_type {
                    OpType::Put => {
                        if let Some(doc_data) = &prepared_op.doc_data {
                            if data_table.insert(&prepared_op.id, doc_data.as_slice())?.is_none() {
                                new_documents_count += 1;
                            }
                            
                            // Deserialize doc_data to get json_blob for Tantivy indexing
                            if let Ok(stored_doc) = serde_json::from_slice::<StoredDoc>(doc_data) {
                                if let Some(json_blob) = stored_doc.json_blob {
                                    // Build Tantivy document (similar to apply_batch logic)
                                    let mut tantivy_doc = doc!(
                                        fields.id => prepared_op.id.as_str(),
                                        fields.seq => seq_id
                                    );
                                    
                                    // Index schema-defined fields from json_blob
                                    // (Full field indexing logic from apply_batch)
                                    
                                    tantivy_ops.push(("add", tantivy_doc, prepared_op.id.clone()));
                                }
                            }
                        }
                    }
                    OpType::Delete => {
                        data_table.remove(&prepared_op.id)?;
                        tantivy_ops.push(("delete", doc!(), prepared_op.id.clone()));
                    }
                }
            }
        }
        
        write_txn.commit()?;
        
        // Apply Tantivy operations after redb commit
        let mut writer = writer_arc.lock().unwrap();
        for (op_type, tantivy_doc, id) in tantivy_ops {
            match op_type {
                "add" => writer.add_document(tantivy_doc)?,
                "delete" => {
                    let id_term = tantivy::Term::from_field_text(fields.id, &id);
                    writer.delete_term(id_term);
                }
                _ => {}
            }
        }
        writer.commit()?;
        
        Ok(((start_seq..start_seq + prepared_ops.len() as u64).collect(), new_documents_count))
    }
}

    /// Calculate deterministic fingerprint from field names
    pub fn calculate_field_fingerprint(&self) -> u64 {
        let mut field_names: Vec<&String> = self.fields.keys().collect();
        field_names.sort();  // Guarantee order (JSON objects are unordered)
        
        let mut hasher = xxhash_rust::xxh3::Xxh3::new();
        for name in field_names {
            hasher.update(name.as_bytes());
        }
        hasher.finish()
    }

        let mut changed = false;
        
        for (name, def) in new_fields {
            if !self.fields.contains_key(&name) {
                self.fields.insert(name, def);
                changed = true;
            }
        }
        
        if changed {
            self.version += 1;
            self.field_fingerprint = self.calculate_field_fingerprint();
            self.updated_at = chrono::Utc::now().timestamp();
        }
        
        changed
    }

        } else {

        if !self.fields.contains_key(&field_name) {
            return Err(format!("Field '{}' does not exist in schema", field_name));
        }
        self.routing_field_name = field_name;
        Ok(())
    }
    
    /// Auto-detect and set routing field using priority algorithm
    pub fn auto_detect_routing_field(&mut self) {
        // Priority: id → hash fields → suffix → substring → first field
        if self.fields.contains_key("id") {
            self.routing_field_name = "id".to_string();
            return;
        }
        
        const HASH_ALGORITHMS: &[&str] = &["sha256", "sha1", "md5"];
        for hash_name in HASH_ALGORITHMS {
            if self.fields.contains_key(*hash_name) {
                self.routing_field_name = hash_name.to_string();
                return;
            }
        }
        
        // Suffix match
        for field_name in self.fields.keys() {
            let lower = field_name.to_lowercase();
            if lower.ends_with("id") || lower.ends_with("_id") {
                self.routing_field_name = field_name.clone();
                return;
            }
        }
        
        // Substring match
        for field_name in self.fields.keys() {
            if field_name.to_lowercase().contains("id") {
                self.routing_field_name = field_name.clone();
                return;
            }
        }
        
        // Fallback: first field in sorted order
        let mut sorted_fields: Vec<&String> = self.fields.keys().collect();
        sorted_fields.sort();
        if let Some(first_field) = sorted_fields.first() {
            self.routing_field_name = first_field.to_string();
        } else {
            self.routing_field_name = "id".to_string(); // Ultimate fallback
        }
    }

/// Auto-detect routing field from document (for schema creation)
pub fn detect_routing_field(doc: &JsonValue) -> Option<IdFieldDetection> {
    let obj = doc.as_object()?;
    
    const HASH_ALGORITHMS: &[&str] = &["sha256", "sha1", "md5"];
    
    // Priority 1: Exact match for "id"
    if let Some(value) = obj.get("id") {
        if let Some(val_str) = extract_string_value(value) {
            return Some(IdFieldDetection {
                field_name: "id".to_string(),
                field_value: val_str,
                is_shadow: false,
            });
        }
    }
    
    // Priority 2: Hash algorithm fields
    for hash_name in HASH_ALGORITHMS {
        if let Some(value) = obj.get(*hash_name) {
            if let Some(val_str) = extract_string_value(value) {
                return Some(IdFieldDetection {
                    field_name: hash_name.to_string(),
                    field_value: val_str,
                    is_shadow: true,
                });
            }
        }
    }
    
    // Priority 3: Suffix-based matches
    for (field_name, value) in obj {
        let lower = field_name.to_lowercase();
        if lower.ends_with("id") || lower.ends_with("_id") {
            if let Some(val_str) = extract_string_value(value) {
                return Some(IdFieldDetection {
                    field_name: field_name.clone(),
                    field_value: val_str,
                    is_shadow: field_name.to_lowercase() != "id",
                });
            }
        }
    }
    
    // Priority 4: Substring match
    for (field_name, value) in obj {
        if field_name.to_lowercase().contains("id") {
            if let Some(val_str) = extract_string_value(value) {
                return Some(IdFieldDetection {
                    field_name: field_name.clone(),
                    field_value: val_str,
                    is_shadow: field_name.to_lowercase() != "id",
                });
            }
        }
    }
    
    // Priority 5: First field in sorted list
    let mut sorted_fields: Vec<(&String, &JsonValue)> = obj.iter().collect();
    sorted_fields.sort_by_key(|(name, _)| *name);
    
    if let Some((field_name, value)) = sorted_fields.first() {
        if let Some(val_str) = extract_string_value(value) {

                field_name: field_name.to_string(),
                field_value: val_str,
                is_shadow: field_name.to_lowercase() != "id",
            });
        }
    }
    
    None
}

/// Optimized prepared operation with single serialization
#[derive(Debug)]
pub struct PreparedOp {
    pub id: String,
    pub json_bytes: Vec<u8>,  // Serialized ONCE, used for both WAL and data
    pub op_type: OpType,
    pub json_value: Option<JsonValue>,  // Keep parsed for Tantivy indexing

#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum OpType {
    Put,
    Delete,
}

impl HybridStore {
    /// Prepare operations with single serialization
    pub fn prepare_batch_ops(
        &self,
        index: &str,
        ops: Vec<WalOp>,
    ) -> Result<Vec<PreparedOp>, StoreError> {
        let schema = self.get_schema_cached(index)?
            .unwrap_or_else(|| Arc::new(IndexSchema::default()));
        
        let mut prepared = Vec::with_capacity(ops.len());
        
        for op in ops {
            match op {
                WalOp::Put { id, json_blob } => {
                    if let Some(json_val) = json_blob {
                        let filtered_json = filter_shadow_fields_owned(json_val, &schema);
                        
                        // ✅ SERIALIZE ONCE
                        let json_bytes = serde_json::to_vec(&filtered_json)?;
                        
                        prepared.push(PreparedOp {
                            id,
                            json_bytes,
                            op_type: OpType::Put,
                            json_value: Some(filtered_json),
                        });
                    }
                }
                WalOp::Delete { id } => {
                    prepared.push(PreparedOp {
                        id,
                        json_bytes: Vec::new(),
                        op_type: OpType::Delete,
                        json_value: None,
                    });
                }
            }
        }
        
        Ok(prepared)
    }
    
    /// Apply prepared operations (reuse serialized bytes)
    pub fn apply_prepared_batch(
        &self,
        index: &str,
        prepared_ops: Vec<PreparedOp>,
    ) -> Result<(Vec<u64>, usize), StoreError> {
        let start_seq = self.get_next_seq_batch(index, prepared_ops.len())?;
        
        let mut write_txn = self.kv.begin_write()?;
        let mut new_documents_count = 0;
        let mut tantivy_ops = Vec::new();
        
        let data_table_name = format!("data_{}", index);
        let wal_table_name = format!("wal_{}", index);
        let data_table_def = TableDefinition::<&str, &[u8]>::new(&data_table_name);
        let wal_table_def = TableDefinition::<u64, &[u8]>::new(&wal_table_name);
        
        {
            let mut wal_table = write_txn.open_table(wal_table_def)?;
            let mut data_table = write_txn.open_table(data_table_def)?;
            
            for (prepared_op, seq_id) in prepared_ops.iter().zip(start_seq..) {
                match prepared_op.op_type {
                    OpType::Put => {
                        // ✅ REUSE BYTES: Write same json_bytes to both tables
                        wal_table.insert(seq_id, prepared_op.json_bytes.as_slice())?;
                        
                        if data_table.insert(&prepared_op.id, prepared_op.json_bytes.as_slice())?.is_none() {
                            new_documents_count += 1;
                        }
                        
                        if let Some(ref json_value) = prepared_op.json_value {
                            tantivy_ops.push((seq_id, json_value.clone(), prepared_op.id.clone()));
                        }
                    }
                    OpType::Delete => {
                        wal_table.insert(seq_id, b"DELETE")?;
                        data_table.remove(&prepared_op.id)?;
                        tantivy_ops.push((seq_id, JsonValue::Null, prepared_op.id.clone()));
                    }
                }
            }
        }
        
        write_txn.commit()?;
        self.index_tantivy_batch(index, tantivy_ops)?;
        
        let seq_ids: Vec<u64> = (start_seq..start_seq + prepared_ops.len() as u64).collect();
        Ok((seq_ids, new_documents_count))
    }
}

    pub fn process_document_single_pass(
        &self,
        doc: JsonValue,
        schema: &IndexSchema,
    ) -> Result<ProcessedDocument, StoreError> {

        let mut tantivy_fields = Vec::new();

                    continue;
                }

                let field_type = infer_field_type(&field_value);
                fields.insert(field_name.clone(), field_type);
                
                // Extract Tantivy field value (indexing)
                if let Some(field_def) = schema.fields.get(&field_name) {
                    if field_def.indexed {
                        if let Some(tantivy_value) = self.extract_tantivy_value(
                            &field_value,
                            &field_def.field_type,
                        )? {
                            tantivy_fields.push((field_name.clone(), tantivy_value));
                        }
                    }
                }
                
                filtered_obj.insert(field_name, field_value);
            }
            
            JsonValue::Object(filtered_obj)
        } else {
            doc
        };
        
        let json_bytes = serde_json::to_vec(&filtered_json)?;
        
        Ok(ProcessedDocument {
            filtered_json,
            json_bytes,
            fields,
            tantivy_fields,
        })
    }

        &self,
        value: &JsonValue,
        field_type: &TantivyFieldType,

                if let Some(s) = value.as_str() {
                    Ok(Some(TantivyFieldValue::Text(s.to_string())))
                } else {
                    Ok(Some(TantivyFieldValue::Text(serde_json::to_string(value)?)))
                }
            }
            TantivyFieldType::I64 => Ok(value.as_i64().map(TantivyFieldValue::I64)),
            TantivyFieldType::U64 => Ok(value.as_u64().map(TantivyFieldValue::U64)),
            TantivyFieldType::F64 => Ok(value.as_f64().map(TantivyFieldValue::F64)),
            TantivyFieldType::Boolean => Ok(value.as_bool().map(TantivyFieldValue::Bool)),

// node_orchestrator.rs

/// Extract field names from documents (lightweight operation)
fn extract_field_names(docs: &[DocPayload]) -> HashSet<String> {
    let mut field_names = HashSet::new();
    for doc in docs {
        if let Some(obj) = doc.doc.as_object() {
            field_names.extend(obj.keys().cloned());
        }
    }
    field_names
}

/// Calculate fingerprint from field names
fn calculate_fingerprint(field_names: &HashSet<String>) -> u64 {
    let mut sorted_names: Vec<&String> = field_names.iter().collect();
    sorted_names.sort();
    
    let mut hasher = xxhash_rust::xxh3::Xxh3::new();
    for name in sorted_names {
        hasher.update(name.as_bytes());
    }
    hasher.finish()
}

impl NodeOrchestrator {
    /// Fully optimized bulk write pipeline with schema-based routing
    async fn orch_bulk_write_optimized(
        &self,
        index: &str,
        docs: Vec<DocPayload>,
    ) -> Result<JsonValue, OrchestratorError> {
        // Step 1: Extract field names and calculate fingerprint
        let batch_field_names = extract_field_names(&docs);
        let batch_fingerprint = calculate_fingerprint(&batch_field_names);
        
        // Step 2: Get schema by fingerprint (instant cache lookup)
        let schema = if let Some(schema) = self.schema_cache.get_by_fingerprint(batch_fingerprint) {
            schema
        } else {
            // Cache miss - evolve or create schema
            self.get_or_evolve_schema(index, &docs, batch_fingerprint).await?
        };
        
        // Step 3: Extract routing field name from schema
        let routing_field = schema.get_routing_field();
        
        // Step 4: Group documents by shard using schema routing field
        let mut local_batches: HashMap<Uuid, Vec<DocPayload>> = HashMap::new();
        
        for doc in docs {
            // Extract routing value from document using schema's routing field
            let routing_value = extract_routing_value(
                &doc.doc,
                routing_field,
            ).unwrap_or_else(|| {
                // Fallback: use id or hash entire document
                doc.id.clone()
            });
            
            // Route to shard using consistent hash ring
            if let Some(shard_id) = self.routing_ring.get_owner(&routing_value) {
                local_batches.entry(shard_id).or_default().push(doc);
            }
        }
        
        // Step 5: Process batches with single-pass processing + single serialization
        self.parallel_local_shard_processing(index, local_batches).await
    }
    
    /// Create schema from documents with auto-detection of routing field
    async fn create_schema_from_docs(
        &self,
        index: &str,
        docs: &[DocPayload],
    ) -> Result<Arc<IndexSchema>, OrchestratorError> {
        let mut schema = IndexSchema::default();
        
        // Extract fields from documents
        for doc in docs {
            if let Some(obj) = doc.doc.as_object() {
                for (field_name, field_value) in obj {
                    schema.evolve_field(field_name.clone(), field_value);
                }
            }
        }
        
        // Auto-detect routing field using priority algorithm
        schema.auto_detect_routing_field();
        
        // Set version and fingerprint
        schema.version = 1;
        schema.field_fingerprint = schema.calculate_field_fingerprint();
        schema.created_at = chrono::Utc::now().timestamp();
        schema.updated_at = schema.created_at;
        
        Ok(Arc::new(schema))
    }
}

impl RouterActor {
    /// Enhanced route_and_handle with direct write routing
    #[cfg_attr(not(test), allow(dead_code))]
    pub async fn route_and_handle(
        &self,
        op: ClientOp,
        routing_key: Option<String>,
        operation_type: OperationType,
    ) -> Result<JsonValue, OrchestratorError> {
        // Write operations: Route directly to shards
        if matches!(operation_type, OperationType::Write) {
            return self.handle_write_operation(op, routing_key).await;
        }
        
        // Metadata operations: Route to orchestrator
        if self.is_metadata_operation(&op) {
            return self.orchestrator.ask(op).await;
        }
        
        // Read operations: Route to orchestrator (async)
        return self.orchestrator.ask(op).await;
    }
    
    /// Handle write operations directly without orchestrator involvement
    async fn handle_write_operation(
        &self,
        op: ClientOp,
        routing_key: Option<String>,
    ) -> Result<JsonValue, OrchestratorError> {
        match op {
            ClientOp::Write { index, id, routing_key: _, doc } => {
                self.route_single_write(&index, id, doc, routing_key).await
            }
            ClientOp::BulkWrite { index, docs } => {
                self.route_bulk_write(&index, docs, routing_key).await
            }
            _ => Err(OrchestratorError::Io(std::io::Error::new(
                std::io::ErrorKind::InvalidInput,
                "Not a write operation",
            ))),
        }
    }
    
    /// Route single write operation directly to shard
    async fn route_single_write(
        &self,
        index: &str,
        id: String,
        doc: JsonValue,
        explicit_routing_key: Option<String>,
    ) -> Result<JsonValue, OrchestratorError> {
        // Get routing decision from coordinator
        let decision = self.coordinator
            .ask(RouteOperation {
                routing_key: explicit_routing_key,
                operation_type: OperationType::Write,
            })
            .await?;
        
        match decision {
            RoutingDecision::Local { target_shards } => {
                // Route to local shard directly
                let shard_id = target_shards.first().ok_or_else(|| {
                    OrchestratorError::Io(std::io::Error::new(
                        std::io::ErrorKind::NotFound,
                        "No local shard available",
                    ))
                })?;
                
                // Get shard reference from orchestrator (metadata operation)
                let shard_info = self.orchestrator.ask(GetShardInfo { shard_id: *shard_id }).await?;
                
                // Execute write directly on shard
                let write_req = WriteRequest {
                    index: index.to_string(),
                    routing_key: explicit_routing_key.unwrap_or_else(|| id.clone()),
                    doc,
                };
                
                // Use MicroshardActor directly
                let shard_actor = shard_info.actor_ref;
                let result = shard_actor.ask(write_req).await?;
                
                Ok(serde_json::json!({
                    "id": id,
                    "result": "created",
                    "version": result.sequence,
                    "shard_id": shard_id.to_string()
                }))
            }
            RoutingDecision::Remote { target_nodes } => {
                // Route to remote node
                self.route_write_remote(&index, id, doc, explicit_routing_key, target_nodes).await
            }
            RoutingDecision::Broadcast { target_nodes } => {
                // Broadcast write (rare case)
                self.route_write_broadcast(&index, id, doc, explicit_routing_key, target_nodes).await
            }
        }
    }
    
    /// Route bulk write operation directly to shards
    async fn route_bulk_write(
        &self,
        index: &str,
        docs: Vec<DocPayload>,
        explicit_routing_key: Option<String>,
    ) -> Result<JsonValue, OrchestratorError> {
        // Get schema for routing field detection
        let schema = self.orchestrator.ask(GetSchema { index: index.to_string() }).await?;
        
        // Extract routing field from schema
        let routing_field = schema.get_routing_field();
        
        // Group documents by routing value
        let mut routing_groups: HashMap<String, Vec<DocPayload>> = HashMap::new();
        
        for doc in docs {
            let routing_value = extract_routing_value(&doc.doc, routing_field)
                .unwrap_or_else(|| doc.id.clone());
            
            routing_groups.entry(routing_value).or_default().push(doc);
        }
        
        // Route each group
        let mut results = Vec::new();
        let mut total_written = 0u64;
        
        for (routing_value, group_docs) in routing_groups {
            let decision = self.coordinator
                .ask(RouteOperation {
                    routing_key: Some(routing_value),
                    operation_type: OperationType::Write,
                })
                .await?;
            
            match decision {
                RoutingDecision::Local { target_shards } => {
                    let batch_result = self.execute_local_bulk_write(index, group_docs, target_shards).await?;
                    total_written += batch_result.items_written;
                    results.push(batch_result);
                }
                RoutingDecision::Remote { target_nodes } => {
                    let batch_result = self.route_bulk_write_remote(index, group_docs, target_nodes).await?;
                    total_written += batch_result.items_written;
                    results.push(batch_result);
                }
                _ => {
                    // Handle broadcast case for bulk writes
                    warn!("Bulk write broadcast not implemented, falling back to orchestrator");
                    return self.orchestrator.ask(ClientOp::BulkWrite { index: index.to_string(), docs }).await;
                }
            }
        }
        
        Ok(serde_json::json!({
            "items_written": total_written,
            "results": results
        }))
    }
    
    /// Check if operation is metadata-related
    fn is_metadata_operation(&self, op: &ClientOp) -> bool {
        matches!(op, 
            ClientOp::CreateConfig { .. } |
            ClientOp::GetConfig { .. } |
            ClientOp::ListIndexes { .. } |
            ClientOp::CreateIndex { .. } |
            ClientOp::DeleteIndex { .. }
        )
    }
}
